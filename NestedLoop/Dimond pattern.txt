import java.util.Scanner;

class Pattern{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int num = sc.nextInt();
        for(int i=1;i<=num;i=i+2){
            for(int j=1;j<=i;j++){
                System.out.print("*");
            }
            System.out.println();
        }
        for(int i=1;i<=num;i=i+2){
            for(int j=num-2;j>=i;j--){
                System.out.print("*");
            }
            System.out.println();
        }
    }
}
