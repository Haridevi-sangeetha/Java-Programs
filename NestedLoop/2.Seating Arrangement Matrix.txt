import java.util.Scanner;

class SeatingArrangement{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int row = sc.nextInt();
        int seatsPerRow=sc.nextInt();
        for(int i=1;i<=row;i++){
            for(int j=1;j<=seatsPerRow;j++){
                System.out.print("Seat-"+j+" ");
            }
            System.out.println();
        }
    }
}
