import java.util.Scanner;

public class ParkingLotGrid {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int rows = sc.nextInt();
        int cols = sc.nextInt();
        int occupiedSpots = sc.nextInt();

        boolean[][] lot = new boolean[rows][cols];

        for (int i = 0; i < occupiedSpots; i++) {
            int r = sc.nextInt();
            int c = sc.nextInt();
            lot[r][c] = true;
        }

        System.out.println("Parking Lot Grid:");
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (lot[i][j])
                    System.out.print("[O] ");
                else
                    System.out.print("[ ] ");
            }
            System.out.println();
        }

        int total = rows * cols;
        int vacant = total - occupiedSpots;
        double rate = (occupiedSpots * 100.0) / total;

        System.out.println();
        System.out.println("Total Spots: " + total);
        System.out.println("Occupied: " + occupiedSpots);
        System.out.println("Vacant: " + vacant);
        System.out.printf("Occupancy Rate: %.2f%%", rate);

        sc.close();
    }
}
