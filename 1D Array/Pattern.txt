import java.util.Scanner;

class Pattern{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int num = sc.nextInt();
        for(int i=0;i<num;i++){
            char a = 'A';
            for(int j=0;j<=i;j++){
                if(j%2==0){
                    System.out.printf("%c",a+j);
                    System.out.print(" ");
                }
                else if(j%2==1){
                    System.out.print(i+1);
                    System.out.print(" ");
                }
            }
            System.out.println("");
        }
    }
}
