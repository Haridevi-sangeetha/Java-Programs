import java.util.*;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long[] arr = new long[n];

        long sum = 0;
        long max = 0;
        int maxIndex = 0;

        for(int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
            sum += arr[i];

            if(arr[i] > max) {
                max = arr[i];
                maxIndex = i + 1;
            }
        }

        double avg = (double)sum / n;
        double threshold = avg * 1.5;

        int count = 0;
        ArrayList<Integer> days = new ArrayList<>();

        for(int i = 0; i < n; i++) {
            if(arr[i] > threshold) {
                count++;
                days.add(i + 1);
            }
        }

        System.out.println("Trading Days Analyzed: " + n);
        System.out.println("Total Volume: " + sum);
        System.out.printf("Average Daily Volume: %.0f\n", avg);
        System.out.println("Peak Volume Day: Day " + maxIndex + " (" + max + ")");
        System.out.println("High Volume Days (>150% avg): " + count);
        System.out.println("Breakout Signal Days: " + days);
    }
}
