import java.util.*;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] arr = new int[n];

        int sum = 0;
        int max = 0;
        int maxIndex = 0;

        for(int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            sum += arr[i];

            if(arr[i] > max) {
                max = arr[i];
                maxIndex = i + 1;
            }
        }

        int target = sc.nextInt();

        int below = 0;
        for(int i = 0; i < n; i++) {
            if(arr[i] < target)
                below++;
        }

        double avg = (double)sum / n;
        double goal = ((n - below) * 100.0) / n;

        System.out.println("Days Tracked: " + n);
        System.out.println("Total Calories Burned: " + sum);
        System.out.printf("Average Daily Burn: %.2f\n", avg);
        System.out.println("Peak Burn Day: Day " + maxIndex + " (" + max + " calories)");
        System.out.println("Days Below Target: " + below);
        System.out.printf("Goal Achievement: %.2f%%\n", goal);
    }
}
