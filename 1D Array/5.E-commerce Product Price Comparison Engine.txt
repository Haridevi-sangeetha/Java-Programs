import java.util.Scanner;

class ECommerce{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int sellers = sc.nextInt();
        double [] prices = new double[sellers];
        for(int i=0;i<sellers;i++){
            prices[i] = sc.nextDouble();
        }
        double Max = prices[0];
        double Min = prices[0];
        
        for(int i=0;i<sellers;i++){
            if(prices[i]>Max){
                Max = prices[i];
            }
            if(prices[i]<Min){
                Min = prices[i];
            }
        }
        System.out.println("Number of Sellers: "+sellers);
        System.out.printf("Lowest Price: $"+"%.2f%n",Min);
        System.out.printf("Highest Price: $"+"%.2f%n",Max);
        double difference = Max - Min;
        System.out.printf("Price Difference: $"+"%.2f%n",difference);
        double Saving = (difference / Max)*100;
        System.out.printf("Savings: "+"%.2f",Saving);
        System.out.print("%");
    }
}
