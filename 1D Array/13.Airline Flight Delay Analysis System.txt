import java.util.Scanner;

class FlightDelay{
    public static void main(String [] args){
        Scanner sc = new Scanner (System.in);
        int flight = sc.nextInt();
        int []delay = new int [flight];
        int sum =0;
        int max = delay[0];
        int OntimeFlight = 0;
        int Compension = 0;
        for(int i=0;i<flight;i++){
            delay[i]= sc.nextInt();
            sum = sum +delay[i];
            if(delay[i]>max){
                max = delay[i];
            }
            if(delay[i]<=15){
                OntimeFlight++;
            }
            if(delay[i]>180){
                Compension++;
            }
        }
        double avg = (double)sum/(double)flight;
        
        System.out.println("Total Flight: "+flight);
        System.out.printf("Average Delay: "+"%.2f",avg);
        System.out.println(" minutes");
        System.out.println("Maximum Delay: "+max+" minutes");
        System.out.println("on-Time Flights: "+OntimeFlight);
        System.out.println("Compension Required: "+Compension);
        
        double percentage = (OntimeFlight/(double)flight)*100;
        
        System.out.printf("On-Time Performance: "+"%.2f%%",percentage);
    }
}
