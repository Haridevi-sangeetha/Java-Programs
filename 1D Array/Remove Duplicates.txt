Remove Duplicates


import java.util.Scanner;

class Main{
    public static void main(String [] args){
        Scanner sc =  new Scanner(System.in);
        int size=sc.nextInt();
        int []arr=new int[size];
        for(int i=0;i<size;i++){
            arr[i]=sc.nextInt();
        }
        int count;
        int num=0;
        System.out.print("[");
        for(int i=0;i<size;i++){
            count=0;
            for(int j=i+1;j<size;j++){
                if(arr[i]==arr[j]){
                    count++;
                }
            }
            num++;
            if(count==0){
                System.out.print(arr[i]);
                if(num<size){
                    System.out.print(", ");
                }
            }
        }
        System.out.print("]");
    }
}
