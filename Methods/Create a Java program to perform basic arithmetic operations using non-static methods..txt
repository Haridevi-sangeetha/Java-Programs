import java.util.*;


// Here the methods are non-static. That means the object creation is mandatory without object creation methods can't accessed . Because non-static methods are belongs to objects;
public class Main{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        ArithmeticOperators obj = new ArithmeticOperators();
        
        System.out.println(obj.add(a,b));
        System.out.println(obj.sub(a,b));
        System.out.println(obj.mul(a,b));
        System.out.println(obj.mod(a,b));
        System.out.println(obj.div(a,b));
    }
}

class ArithmeticOperators{
    public static int add(int a, int b){
        int add = a+b;
        return add;
    } 
    public static int sub(int a, int b){
        int sub = a-b;
        return sub;
    }
    public static int mul(int a, int b){
        int mul = a*b;
        return mul;
    }
    public static int mod (int a,int b){
        int mod = a%b;
        return mod;
    }
    public static float div(int a,int b){
        int div = a/b;
        return div;
    }
}
