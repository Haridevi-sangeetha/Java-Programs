import java.util.Scanner;

class Password{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        int size = sc.nextInt();
        String []arr = new String[size];
        for(int i=0;i<size;i++){
            arr[i] = sc.next();  
        }
        for(int i=0;i<size;i++){
            int len=arr[i].length();
            int UpperCount=0;
            int LowerCount=0;
            int NumCount=0;
            int SpecialCount=0;
            if(len<8){
                System.out.println(arr[i]+" - "+"Weak");
            }
            else if(len>=8){
                for(int j=0;j<len;j++){
                    char ch = arr[i].charAt(j);
                    if(Character.isUpperCase(ch)){
                        UpperCount++;
                    }
                    if(Character.isLowerCase(ch)){
                        LowerCount++;
                    }
                    if(Character.isDigit(ch)){
                        NumCount++;
                    }
                    if((!(Character.isUpperCase(ch)))&&(!(Character.isDigit(ch)))&&(!(Character.isLowerCase(ch)))&&(ch!='.')){
                        SpecialCount++;
                    }
                }
                if(UpperCount>=1 && LowerCount>=1 && NumCount>=1 && SpecialCount>=1){
                    System.out.println(arr[i]+" - "+"Strong");
                }
                else{
                    System.out.println(arr[i]+" - "+"Weak");
                }
            }
        }
    }
}
